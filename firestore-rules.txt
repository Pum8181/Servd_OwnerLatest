// Paste into Firebase Console → Firestore Database → Rules
//
// WARNING: These rules allow anyone with your Firebase web config to read
// and write the entire menu and orders collections. That is acceptable for
// a single-restaurant free pilot with no user accounts, but you MUST tighten
// this before onboarding multiple restaurants:
//   - Use Firebase Auth (custom claims or role-based rules) for owner writes
//   - Restrict customer writes to orders only (menu should be owner-read/write)
//   - Consider App Check to reduce abuse from non-browser clients
//
// Until then, treat owner.html PIN as a convenience gate only — not security.

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    match /menu/{itemId} {
      allow read, write: if true;
    }

    match /orders/{orderId} {
      allow read, write: if true;
    }

    // NEW: backs the staff PIN login overlay in owner_v3.html. Same
    // "convenience gate, not real security" caveat as above — PINs are
    // plain text and readable by anyone with your Firebase web config.
    match /staff/{staffId} {
      allow read, write: if true;
    }

    // NEW: backs the "Request a server" flow (customer bell button +
    // owner Server Requests panel). Same convenience-gate caveat as
    // everything else here.
    match /serverRequests/{requestId} {
      allow read, write: if true;
    }

    // NEW: restaurant-wide settings (currently just the owner-editable
    // spice level names, e.g. Mild/Medium/Hot/Extra Hot) — a single
    // doc (settings/general), not per-item. Same convenience-gate
    // caveat as everything else here.
    match /settings/{docId} {
      allow read, write: if true;
    }
  }
}
